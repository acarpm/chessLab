void loop() {
    digitalWrite(DEC_A0, 0);
    digitalWrite(DEC_A1, 0);
    digitalWrite(DEC_A2, 0);
    digitalWrite(SEL_S0, 0);
    digitalWrite(SEL_S1, 0);
    digitalWrite(SEL_S2, 0);
    delayMicroseconds(10);
    
    int val = analogRead(HALL_DOUT);
    Serial.printf("ADC = %d\n", val);
    delay(200);
}
